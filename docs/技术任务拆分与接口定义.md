### 1.2 技术栈选型

| 层级 | 选型 | 说明 |
|------|------|------|
| 后端框架 | **Fastify** | 比 Express 性能更高，适合高并发 Trace 上报 |
| ORM | **Drizzle ORM** | 类型安全、轻量、迁移友好 |
| 数据库 | **PostgreSQL** + **Redis** | 主库 PG，缓存 Redis |
| 时序数据 | **TimescaleDB** (可选) | 大规模 Trace 存储时使用 |
| 任务队列 | **BullMQ** (Redis) | 异步任务（评测运行、告警、数据回流） |
| 前端框架 | **React 19** + **Vite** | 已有技术栈延续 |
| 状态管理 | **Zustand** | 已有技术栈延续 |
| 图表 | **Recharts** | 已有技术栈延续 |
| SDK 语言 | TypeScript / Python / Go | 已有技术栈延续 |
| CLI | **oclif** (Node.js) | 跨平台 CLI 框架 |
| 协议 | **OpenTelemetry** | 标准化 Trace 格式 |

#### 2.1.2 数据库表结构

```sql
-- prompts 表（Prompt 主表）
CREATE TABLE prompts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  agent_id VARCHAR(100) NOT NULL,
  name VARCHAR(200) NOT NULL,
  description TEXT,
  current_version_id UUID,
  git_repo VARCHAR(500),
  git_branch VARCHAR(100) DEFAULT 'main',
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(project_id, agent_id, name)
);

-- prompt_versions 表（Prompt 版本）
CREATE TABLE prompt_versions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  prompt_id UUID NOT NULL REFERENCES prompts(id) ON DELETE CASCADE,
  version VARCHAR(100) NOT NULL,           -- Git tag 或语义化版本
  content TEXT NOT NULL,
  variables TEXT[] DEFAULT '{}',
  metadata JSONB DEFAULT '{}',              -- { model, temperature, max_tokens }
  
  -- Git 信息
  git_commit_hash VARCHAR(40),
  git_commit_message TEXT,
  git_author VARCHAR(200),
  git_timestamp TIMESTAMPTZ,
  
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(prompt_id, version)
);

-- prompt_experiments 表（A/B 实验）
CREATE TABLE prompt_experiments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  agent_id VARCHAR(100) NOT NULL,
  name VARCHAR(200) NOT NULL,
  status VARCHAR(20) NOT NULL DEFAULT 'draft',  -- draft/running/paused/completed/cancelled
  
  variant_a JSONB NOT NULL,                 -- { prompt_version_id, traffic_percent }
  variant_b JSONB NOT NULL,
  
  primary_metric VARCHAR(50) DEFAULT 'success_rate',
  secondary_metrics TEXT[] DEFAULT '{}',
  
  started_at TIMESTAMPTZ,
  ended_at TIMESTAMPTZ,
  result JSONB,                             -- { winner, confidence_level, improvement_percent }
  
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- experiment_assignments 表（A/B 实验用户分组记录）
CREATE TABLE experiment_assignments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  experiment_id UUID NOT NULL REFERENCES prompt_experiments(id) ON DELETE CASCADE,
  session_id VARCHAR(100) NOT NULL,
  user_id VARCHAR(100),
  variant VARCHAR(10) NOT NULL,             -- 'a' | 'b'
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(experiment_id, session_id)
);

-- Indexes
CREATE INDEX idx_prompts_project ON prompts(project_id);
CREATE INDEX idx_prompt_versions_prompt ON prompt_versions(prompt_id);
CREATE INDEX idx_experiments_project ON prompt_experiments(project_id, status);
CREATE INDEX idx_exp_assignments_experiment ON experiment_assignments(experiment_id);
```

```typescript
// TypeScript SDK
interface PromptConfig {
  agentId: string;
  version?: string;             // 默认最新
  gitTrack?: boolean;           // 是否追踪 Git 版本
}

class AgentMonitor {
  // 注册 Prompt
  registerPrompt(config: {
    agentId: string;
    content: string;
    version?: string;
    metadata?: { model?: string; temperature?: number; max_tokens?: number };
  }): PromptHandle;

  // 从文件加载 Prompt（自动追踪 Git）
  loadPrompt(filePath: string, config?: PromptConfig): Promise<Prompt>;

  // A/B 实验分流
  selectPromptVariant(config: {
    agentId: string;
    sessionId: string;
    userId?: string;
  }): Promise<{ variant: 'a' | 'b'; content: string; version: string }>;

  traceLLM(params: {
    model: string;
    prompt?: string;
    promptVersion?: string;       // 新增
    input: object;
    output: object;
    latencyMs: number;
    usage?: { prompt_tokens: number; completion_tokens: number };
  }): void;
}
```

```sql
-- eval_datasets 表（评测数据集）
CREATE TABLE eval_datasets (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  name VARCHAR(200) NOT NULL,
  description TEXT,
  tags TEXT[] DEFAULT '{}',
  source VARCHAR(50) DEFAULT 'manual',     -- manual / trace_reflow / imported
  case_count INTEGER DEFAULT 0,
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- eval_cases 表（测试用例）
CREATE TABLE eval_cases (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  dataset_id UUID NOT NULL REFERENCES eval_datasets(id) ON DELETE CASCADE,
  name VARCHAR(200),
  input JSONB NOT NULL,
  expected_output TEXT,
  expected_schema JSONB,
  evaluation_criteria JSONB,               -- { type, config }
  tags TEXT[] DEFAULT '{}',
  source_trace_id UUID,
  metadata JSONB,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- evaluators 表（评估器定义）
CREATE TABLE evaluators (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  name VARCHAR(200) NOT NULL,
  type VARCHAR(50) NOT NULL,               -- exact_match / contains / json_schema / regex / embedding_similarity / llm_judge / custom
  config JSONB NOT NULL,
  description TEXT,
  is_preset BOOLEAN DEFAULT FALSE,         -- 系统预置
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- eval_runs 表（评测运行）
CREATE TABLE eval_runs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  experiment_id UUID,                      -- 关联 Prompt 实验
  dataset_id UUID NOT NULL REFERENCES eval_datasets(id),
  target_type VARCHAR(50) NOT NULL,        -- prompt / agent
  target_id VARCHAR(100) NOT NULL,
  target_version VARCHAR(100),
  
  status VARCHAR(20) NOT NULL DEFAULT 'queued',  -- queued/running/completed/failed/cancelled
  progress INTEGER DEFAULT 0,              -- 0-100
  
  started_at TIMESTAMPTZ,
  completed_at TIMESTAMPTZ,
  
  summary JSONB,                           -- { total_cases, passed, failed, pass_rate, avg_latency_ms, total_cost }
  dimension_scores JSONB,                  -- { exact_match: 95, relevance: 88, ... }
  baseline_run_id UUID REFERENCES eval_runs(id),
  
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- eval_case_results 表（评测结果详情）
CREATE TABLE eval_case_results (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  run_id UUID NOT NULL REFERENCES eval_runs(id) ON DELETE CASCADE,
  case_id UUID NOT NULL REFERENCES eval_cases(id),
  status VARCHAR(20) NOT NULL,             -- passed / failed / error / skipped
  actual_output TEXT,
  latency_ms INTEGER,
  scores JSONB,                            -- { exact_match: 100, relevance: 85 }
  failure_reasons TEXT[],
  trace_id UUID,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- auto_eval_tasks 表（Trace 自动评测任务）
CREATE TABLE auto_eval_tasks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  name VARCHAR(200) NOT NULL,
  
  -- 筛选条件
  filters JSONB NOT NULL,                  -- { agent_ids, environments, status, time_range }
  
  evaluator_ids TEXT[] NOT NULL,
  sampling_rate DECIMAL(3,2) DEFAULT 1.0,  -- 采样率
  schedule VARCHAR(50) DEFAULT 'realtime', -- realtime / hourly / daily
  
  -- 告警配置
  alert_threshold JSONB,                   -- { pass_rate_min: 80 }
  
  is_active BOOLEAN DEFAULT TRUE,
  last_run_at TIMESTAMPTZ,
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Indexes
CREATE INDEX idx_eval_datasets_project ON eval_datasets(project_id);
CREATE INDEX idx_eval_cases_dataset ON eval_cases(dataset_id);
CREATE INDEX idx_eval_runs_project ON eval_runs(project_id, status);
CREATE INDEX idx_eval_runs_dataset ON eval_runs(dataset_id);
CREATE INDEX idx_eval_results_run ON eval_case_results(run_id);
CREATE INDEX idx_eval_results_case ON eval_case_results(case_id);
```

#### 2.7.3 SDK Auto-instrumentation 实现

```typescript
// TypeScript 实现原理
function autoInstrument(options?: AutoInstrumentOptions): void {
  const { providers = ['openai'], sampleRate = 1.0 } = options || {};
  
  // 拦截 OpenAI SDK
  if (providers.includes('openai')) {
    patchOpenAI();
  }
  
  // 拦截 Anthropic SDK
  if (providers.includes('anthropic')) {
    patchAnthropic();
  }
  
  // 拦截 LangChain
  if (providers.includes('langchain')) {
    patchLangChain();
  }
}

function patchOpenAI() {
  const OriginalOpenAI = require('openai').OpenAI;
  
  class PatchedOpenAI extends OriginalOpenAI {
    chat = {
      ...OriginalOpenAI.prototype.chat,
      completions: {
        create: async (...args: any[]) => {
          const trace = monitor.startTrace('openai.chat.completions');
          const start = Date.now();
          try {
            const result = await OriginalOpenAI.prototype.chat.completions.create.apply(this, args);
            monitor.traceLLM({ traceContext: trace, model: args[0].model, ... });
            return result;
          } catch (error) {
            monitor.endTrace(trace, 'error', error.message);
            throw error;
          }
        }
      }
    };
  }
  
  require.cache[require.resolve('openai')].exports.OpenAI = PatchedOpenAI;
}
```
